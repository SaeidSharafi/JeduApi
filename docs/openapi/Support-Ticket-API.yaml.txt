openapi: 3.0.3
info:
  title: JeduShop Support Ticket API
  description: >-
    Minimal headless support-ticket contract. Authenticated customers create
    and view their own tickets. Staff list, inspect, answer, and close tickets.
    Customer identity comes from the authenticated user; name and phone are not
    accepted as ticket creation fields.
  version: 0.1.0
servers:
  - url: /api/v1
tags:
  - name: Shop / Support Tickets
  - name: Admin / Support Tickets
paths:
  /shop/support-tickets/departments:
    get:
      tags: [Shop / Support Tickets]
      summary: List departments available for support tickets
      operationId: listSupportTicketDepartments
      security:
        - sanctum: []
      responses:
        '200': { $ref: '#/components/responses/DepartmentCollection' }
        '401': { $ref: '#/components/responses/Unauthenticated' }

  /shop/support-tickets:
    get:
      tags: [Shop / Support Tickets]
      summary: List the authenticated customer’s tickets
      operationId: listCustomerSupportTickets
      security:
        - sanctum: []
      parameters:
        - { $ref: '#/components/parameters/Page' }
        - { $ref: '#/components/parameters/PerPage' }
      responses:
        '200': { $ref: '#/components/responses/SupportTicketCollection' }
        '401': { $ref: '#/components/responses/Unauthenticated' }
    post:
      tags: [Shop / Support Tickets]
      summary: Create a support ticket
      operationId: createSupportTicket
      security:
        - sanctum: []
      requestBody:
        required: true
        content:
          application/json:
            schema: { $ref: '#/components/schemas/SupportTicketCreate' }
            example:
              department_id: 2
              title: I cannot access my course
              message: I paid for the course but it is not visible in my account.
      responses:
        '201': { $ref: '#/components/responses/SupportTicket' }
        '401': { $ref: '#/components/responses/Unauthenticated' }
        '422': { $ref: '#/components/responses/ValidationError' }

  /shop/support-tickets/{ticket}:
    parameters:
      - { $ref: '#/components/parameters/TicketUuid' }
    get:
      tags: [Shop / Support Tickets]
      summary: Get one of the authenticated customer’s tickets
      operationId: getCustomerSupportTicket
      security:
        - sanctum: []
      responses:
        '200': { $ref: '#/components/responses/SupportTicket' }
        '401': { $ref: '#/components/responses/Unauthenticated' }
        '404': { $ref: '#/components/responses/NotFound' }

  /admin/support-tickets:
    get:
      tags: [Admin / Support Tickets]
      summary: List support tickets
      operationId: listAdminSupportTickets
      security:
        - sanctum: []
      parameters:
        - name: status
          in: query
          schema: { $ref: '#/components/schemas/SupportTicketStatus' }
        - name: department_id
          in: query
          schema: { type: integer, minimum: 1 }
        - name: q
          in: query
          description: Searches title, message, and related customer name or phone.
          schema: { type: string }
        - { $ref: '#/components/parameters/Page' }
        - { $ref: '#/components/parameters/PerPage' }
      responses:
        '200': { $ref: '#/components/responses/SupportTicketAdminCollection' }
        '401': { $ref: '#/components/responses/Unauthenticated' }
        '403': { $ref: '#/components/responses/Forbidden' }

  /admin/support-tickets/{ticket}:
    parameters:
      - { $ref: '#/components/parameters/TicketId' }
    get:
      tags: [Admin / Support Tickets]
      summary: Inspect a support ticket
      operationId: getAdminSupportTicket
      security:
        - sanctum: []
      responses:
        '200': { $ref: '#/components/responses/SupportTicketAdmin' }
        '401': { $ref: '#/components/responses/Unauthenticated' }
        '403': { $ref: '#/components/responses/Forbidden' }
        '404': { $ref: '#/components/responses/NotFound' }

  /admin/support-tickets/{ticket}/answer:
    parameters:
      - { $ref: '#/components/parameters/TicketId' }
    patch:
      tags: [Admin / Support Tickets]
      summary: Answer a support ticket
      operationId: answerSupportTicket
      security:
        - sanctum: []
      requestBody:
        required: true
        content:
          application/json:
            schema: { $ref: '#/components/schemas/SupportTicketAnswer' }
      responses:
        '200': { $ref: '#/components/responses/SupportTicketAdmin' }
        '401': { $ref: '#/components/responses/Unauthenticated' }
        '403': { $ref: '#/components/responses/Forbidden' }
        '404': { $ref: '#/components/responses/NotFound' }
        '422': { $ref: '#/components/responses/ValidationError' }

  /admin/support-tickets/{ticket}/status:
    parameters:
      - { $ref: '#/components/parameters/TicketId' }
    patch:
      tags: [Admin / Support Tickets]
      summary: Change support ticket status
      operationId: changeSupportTicketStatus
      security:
        - sanctum: []
      requestBody:
        required: true
        content:
          application/json:
            schema: { $ref: '#/components/schemas/SupportTicketStatusUpdate' }
      responses:
        '200': { $ref: '#/components/responses/SupportTicketAdmin' }
        '401': { $ref: '#/components/responses/Unauthenticated' }
        '403': { $ref: '#/components/responses/Forbidden' }
        '404': { $ref: '#/components/responses/NotFound' }
        '422': { $ref: '#/components/responses/ValidationError' }

components:
  securitySchemes:
    sanctum:
      type: http
      scheme: bearer
      bearerFormat: token
      description: Sanctum token. The server selects the user or staff guard from the route.
  parameters:
    TicketUuid:
      name: ticket
      in: path
      required: true
      schema: { type: string, format: uuid }
    TicketId:
      name: ticket
      in: path
      required: true
      schema: { type: integer, minimum: 1 }
    Page:
      name: page
      in: query
      schema: { type: integer, minimum: 1, default: 1 }
    PerPage:
      name: per_page
      in: query
      schema: { type: integer, minimum: 1, maximum: 100, default: 15 }
  schemas:
    SupportTicketStatus:
      type: string
      enum: [new, answered, closed]
    Department:
      type: object
      required: [id, name]
      properties:
        id: { type: integer, example: 2 }
        name: { type: string, example: Course Registration }
    CustomerSummary:
      type: object
      required: [id, name, phone]
      properties:
        id: { type: integer, example: 48 }
        name: { type: string, nullable: true, example: Sara Ahmadi }
        phone: { type: string, example: '09121234567' }
    StaffSummary:
      type: object
      required: [id, name]
      properties:
        id: { type: integer, example: 7 }
        name: { type: string, nullable: true, example: Support Admin }
    SupportTicketCreate:
      type: object
      required: [department_id, title, message]
      additionalProperties: false
      properties:
        department_id: { type: integer, minimum: 1, example: 2 }
        title: { type: string, maxLength: 191, example: I cannot access my course }
        message: { type: string, example: I paid for the course but it is not visible in my account. }
    SupportTicketAnswer:
      type: object
      required: [answer]
      additionalProperties: false
      properties:
        answer: { type: string, minLength: 1, maxLength: 10000, example: Your course access has been enabled. }
    SupportTicketStatusUpdate:
      type: object
      required: [status]
      additionalProperties: false
      properties:
        status: { $ref: '#/components/schemas/SupportTicketStatus' }
    SupportTicket:
      type: object
      required: [uuid, department, title, message, answer, status, created_at, updated_at]
      properties:
        uuid: { type: string, format: uuid }
        department: { $ref: '#/components/schemas/Department' }
        title: { type: string }
        message: { type: string }
        answer: { type: string, nullable: true }
        status: { $ref: '#/components/schemas/SupportTicketStatus' }
        answered_at: { type: string, format: date-time, nullable: true }
        closed_at: { type: string, format: date-time, nullable: true }
        created_at: { type: string, format: date-time }
        updated_at: { type: string, format: date-time }
    SupportTicketAdmin:
      allOf:
        - $ref: '#/components/schemas/SupportTicket'
        - type: object
          required: [customer, answered_by]
          properties:
            customer: { $ref: '#/components/schemas/CustomerSummary' }
            answered_by:
              allOf:
                - $ref: '#/components/schemas/StaffSummary'
              nullable: true
    Pagination:
      type: object
      properties:
        current_page: { type: integer, example: 1 }
        last_page: { type: integer, example: 3 }
        per_page: { type: integer, example: 15 }
        total: { type: integer, example: 42 }
    Error:
      type: object
      properties:
        message: { type: string }
        errors: { type: object, additionalProperties: true }
  responses:
    SupportTicket:
      description: Support ticket response
      content:
        application/json:
          schema:
            type: object
            properties:
              data: { $ref: '#/components/schemas/SupportTicket' }
    SupportTicketAdmin:
      description: Admin support ticket response
      content:
        application/json:
          schema:
            type: object
            properties:
              data: { $ref: '#/components/schemas/SupportTicketAdmin' }
    SupportTicketCollection:
      description: Customer support ticket collection
      content:
        application/json:
          schema:
            type: object
            properties:
              data: { type: array, items: { $ref: '#/components/schemas/SupportTicket' } }
              meta: { $ref: '#/components/schemas/Pagination' }
    SupportTicketAdminCollection:
      description: Admin support ticket collection
      content:
        application/json:
          schema:
            type: object
            properties:
              data: { type: array, items: { $ref: '#/components/schemas/SupportTicketAdmin' } }
              meta: { $ref: '#/components/schemas/Pagination' }
    DepartmentCollection:
      description: Department collection
      content:
        application/json:
          schema:
            type: object
            properties:
              data: { type: array, items: { $ref: '#/components/schemas/Department' } }
    Unauthenticated:
      description: Authentication required
      content: { application/json: { schema: { $ref: '#/components/schemas/Error' } } }
    Forbidden:
      description: The authenticated actor lacks permission
      content: { application/json: { schema: { $ref: '#/components/schemas/Error' } } }
    NotFound:
      description: Resource not found
      content: { application/json: { schema: { $ref: '#/components/schemas/Error' } } }
    ValidationError:
      description: Validation failed
      content: { application/json: { schema: { $ref: '#/components/schemas/Error' } } }
