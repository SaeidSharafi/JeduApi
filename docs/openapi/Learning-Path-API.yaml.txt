openapi: 3.0.3
info:
  title: JeduShop Learning Path API
  description: >-
    Proposed contract for Learning Paths. A Learning Path is an ordered
    guide made from existing Productable identities. It is not a Product,
    Product Delivery Option, or purchasable entity. The server base path is
    /api/v1.
  version: 1.0.0
servers:
  - url: /api/v1
tags:
  - name: Admin / Learning Paths
  - name: Shop / Learning Paths
security:
  - sanctum: []
paths:
  /admin/learning-paths:
    get:
      tags: [Admin / Learning Paths]
      summary: List Learning Paths
      operationId: listLearningPaths
      responses:
        '200': { $ref: '#/components/responses/LearningPathCollection' }
    post:
      tags: [Admin / Learning Paths]
      summary: Create a Learning Path
      operationId: createLearningPath
      requestBody: { $ref: '#/components/requestBodies/LearningPathWrite' }
      responses:
        '201': { $ref: '#/components/responses/LearningPath' }
        '422': { $ref: '#/components/responses/ValidationError' }
  /admin/learning-paths/{learning_path}:
    parameters:
      - $ref: '#/components/parameters/LearningPathId'
    get:
      tags: [Admin / Learning Paths]
      summary: Get a Learning Path
      operationId: getLearningPath
      responses:
        '200': { $ref: '#/components/responses/LearningPath' }
        '404': { $ref: '#/components/responses/NotFound' }
    put:
      tags: [Admin / Learning Paths]
      summary: Replace a Learning Path, including published content edits
      operationId: updateLearningPath
      requestBody: { $ref: '#/components/requestBodies/LearningPathWrite' }
      responses:
        '200': { $ref: '#/components/responses/LearningPath' }
        '422': { $ref: '#/components/responses/ValidationError' }
    delete:
      tags: [Admin / Learning Paths]
      summary: Delete a never-published draft Learning Path
      operationId: deleteLearningPathDraft
      responses:
        '204': { description: Draft deleted }
        '422': { $ref: '#/components/responses/ValidationError' }

  /admin/learning-paths/{learning_path}/archive:
    parameters:
      - $ref: '#/components/parameters/LearningPathId'
    post:
      tags: [Admin / Learning Paths]
      summary: Archive a published Learning Path
      operationId: archiveLearningPath
      responses:
        '200': { $ref: '#/components/responses/LearningPath' }
        '403': { description: Missing learning_paths.update permission }
        '404': { $ref: '#/components/responses/NotFound' }
        '422': { $ref: '#/components/responses/ValidationError' }

  /learning-paths:
    get:
      security: []
      tags: [Shop / Learning Paths]
      summary: List published Learning Paths
      operationId: listShopLearningPaths
      parameters:
        - { $ref: '#/components/parameters/PaginationPage' }
        - { $ref: '#/components/parameters/PaginationPerPage' }
      responses:
        '200': { $ref: '#/components/responses/LearningPathCardCollection' }
  /learning-paths/{slug}:
    parameters:
      - name: slug
        in: path
        required: true
        schema: { type: string }
    get:
      security: []
      tags: [Shop / Learning Paths]
      summary: Get a published Learning Path
      operationId: getShopLearningPath
      responses:
        '200': { $ref: '#/components/responses/LearningPathDetail' }
        '404': { $ref: '#/components/responses/NotFound' }

  requestBodies:
    LearningPathWrite:
      required: true
      content:
        application/json:
          schema: { $ref: '#/components/schemas/LearningPathWrite' }

    LearningPathCardCollection: { description: Published Learning Path cards, content: { application/json: { schema: { $ref: '#/components/schemas/LearningPathCardCollectionResponse' } } } }

    NotFound: { description: Resource not found }
    Conflict: { description: State or archival conflict }

        slug: { type: string }
        description: { type: string }

        introduction_description: { type: string }

        steps:
          type: array
          items: { $ref: '#/components/schemas/LearningPathStepWrite' }
    LearningPathStepWrite:
      type: object
      required: [position, productable_type, productable_id, title, description]

        position: { type: integer, minimum: 1 }
        productable_type: { type: string, enum: [course, seminar, digital_asset], description: Stable Productable type. }
        productable_id: { type: integer, description: Existing Productable ID. The API must not create a Productable from this reference. }
        title: { type: string, description: Required path-specific label for the step. }
        description: { type: string, description: Required path-specific description for the step. }
    LearningPath:
      type: object
      required: [id, title, slug, description, introduction_title, introduction_description, conclusion_title, conclusion_description, meta_title, meta_description, meta_keywords, display_order, status, steps, media]

        id: { type: integer }
        title: { type: string }
        slug: { type: string }
        description: { type: string }
        introduction_title: { type: string }
        introduction_description: { type: string }
        conclusion_title: { type: string }
        conclusion_description: { type: string }
        meta_title: { type: string, nullable: true }
        meta_description: { type: string, nullable: true }
        meta_keywords: { type: string, nullable: true }
        display_order: { type: integer, minimum: 0 }
        media: { type: object, additionalProperties: { type: array, items: { type: object } } }
        status: { type: string, enum: [draft, published, archived] }

          items: { $ref: '#/components/schemas/LearningPathStep' }

      required: [id, position, productable_type, productable_id, title, description, productable]

        id: { type: integer }
        position: { type: integer, minimum: 1 }

        from: { type: integer, nullable: true }
        last_page: { type: integer, minimum: 1 }
